 base="/workspace/artifacts/game112/extra-effects/fresh"

 if get_tree().current_scene.scene_file_path=="res://scenes/combat/first_adventure.tscn" and not capture_ability.is_empty():

func fight_new() -> void:
 action("cycle_fight");engine.configure(state)
 for n in 250:
  var battle:=AdventureService.active_battle(state)
  if failures>0 or battle.status!="active":break
  var actor:=engine.current(battle);var command:=engine.enemy_command(battle);var new_power:=false
  if actor.team=="party" and int(battle.budget.main)>0:
   var cls: String=actor.record.advancement_history[0].class_id;var ability:=cls+"-technique-1"
   var targets: Array=battle.units.values().filter(func(target: Dictionary):return target.team=="enemy" and engine.eligible(battle,actor,target,ability))
   if not targets.is_empty() and engine.effects.validate_ability_cost(ability,actor.record.runtime,{"main":int(battle.budget.main),"move":int(battle.budget.move),"reaction":int(actor.reaction)}).ok:
    command={"actor_id":actor.id,"revision":battle.revision,"kind":ability,"target_id":targets[0].id}
    new_power=not used_actions.has(ability)
    if new_power:used_actions.append(ability)
  capture_ability=str(command.kind) if new_power else ""
  if new_power:await capture("actual-"+str(command.kind)+"-before","combat")
  var expected:=engine.command(battle,command);check(expected.ok,"actual new power command")
  action("cycle_command",{"command":command})
  check(AdventureService.active_battle(state).state_hash==expected.battle.state_hash,"same authority and save")
  if new_power:await capture("actual-"+str(command.kind)+"-effect","combat")
 check(AdventureService.active_battle(state).status!="active","actual new-class battle terminal")
